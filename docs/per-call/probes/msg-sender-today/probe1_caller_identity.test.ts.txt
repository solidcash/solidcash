import { describe, it, expect } from 'vitest';
import { compileString } from './direct-compile.js';
import { Contract, StatefulContract } from '../src/index.js';
import type { ConnectedStatefulContract } from '../src/stateful/StatefulContract.js';
import { hexToBin } from '@bitauth/libauth';
import { createTestProvider, setupStatefulUtxos } from './test-util.js';

// PROBE 1: does a callee's `require(msg.sender == <A>)` bind to the ACTUAL caller, or merely to a
// co-spent minting master of the SDK-declared category? call-model.md section 9 states the covenant
// proves "IDENTITY (this master was co-spent), not per-call AUTHORIZATION".

const IX = `
interface IX {
    public int touched;
    function privileged(address expectedSender);
}
`;

// X: privileged() requires msg.sender == a stored authorized address.
const X_SRC = `
pragma solidcash >=0.8.0;
${IX}
contract X() {
    int touched;
    function privileged(address expectedSender) {
        require(msg.sender == expectedSender);
        this.touched = this.touched + 1;
    }
}
`;

// A: the authorized caller. It really calls X.privileged in callX(); it also has an unrelated ping().
const A_SRC = `
pragma solidcash >=0.8.0;
${IX}
contract A(address x) {
    int calls;
    function callX(address expectedSender) {
        IX(x).privileged(expectedSender);
        this.calls = this.calls + 1;
    }
    function ping() { this.calls = this.calls + 1; }
}
`;

describe('probe1: caller identity binding (direct path)', () => {
  async function setup(): Promise<{ scA: ConnectedStatefulContract; scX: ConnectedStatefulContract;
    aLock: Uint8Array; xLock: Uint8Array; }> {
    const provider = createTestProvider();
    const artX = compileString(X_SRC);
    const artA = compileString(A_SRC);
    const x = new Contract(artX, [], { provider });
    const { categoryId: catX } = await setupStatefulUtxos(provider, x, artX.stateLayout!, { touched: 0n },
      { saltContract: true });
    const scX = StatefulContract.withCategory(x, catX);
    const xPinnedForA = hexToBin((x.withCategory(hexToBin(catX)) as any).lockingBytecode);
    const a = new Contract(artA, [xPinnedForA], { provider });
    const { categoryId: catA } = await setupStatefulUtxos(provider, a, artA.stateLayout!, { calls: 0n },
      { saltContract: true });
    const scA = StatefulContract.withCategory(a, catA);
    const aPinnedLock = hexToBin((a.withCategory(hexToBin(catA)) as any).lockingBytecode);
    const xPinnedLock = hexToBin((x.withCategory(hexToBin(catX)) as any).lockingBytecode);
    console.log('PROBE1 addrs', JSON.stringify({
      aUnpinned: (a.lockingBytecode as string).slice(0, 20),
      aPinned: (a.withCategory(hexToBin(catA)) as any).lockingBytecode.slice(0, 20),
    }));
    return { scA, scX, aLock: aPinnedLock, xLock: xPinnedLock };
  }

  // INTENT GAP: A executes ping() (its source contains NO call to X) yet the builder is asked to co-spend
  // X.privileged claiming A as sender. If the VM accepts, msg.sender proves only co-presence, not that A
  // initiated the call (call-model.md section 9).

  it('CONTROL: A really calls X.privileged(addrA) -> accepted', async () => {
    const { scA, scX, aLock } = await setup();
    const tx = await scA.callX(aLock).withExternalCall(scX, 'privileged', [aLock]).send();
    expect(tx.txid).toBeDefined();
    expect((await scX.getState()).touched).toBe(1n);
  });

  it('NEGATIVE: A calls X.privileged(<not A>) -> must be rejected', async () => {
    const { scA, scX, xLock } = await setup();
    let outcome = 'ACCEPTED';
    try {
      await scA.callX(xLock).withExternalCall(scX, 'privileged', [xLock]).send();
    } catch (e: any) { outcome = 'REJECTED ' + String(e?.message ?? e).slice(0, 100).replace(/\n/g, ' '); }
    console.log(`PROBE1 wrong-expected-sender: ${outcome}`);
    expect(true).toBe(true);
  });

  it('INTENT: A runs ping() (no call to X) while co-spending X.privileged(addrA)', async () => {
    const { scA, scX, aLock } = await setup();
    let outcome = 'ACCEPTED (intent gap: presence-based)';
    try {
      await (scA as any).ping().withExternalCall(scX, 'privileged', [aLock]).send();
      const touched = (await scX.getState()).touched;
      outcome += ` touched=${touched}`;
    } catch (e: any) { outcome = 'REJECTED ' + String(e?.message ?? e).slice(0, 140).replace(/\n/g, ' '); }
    console.log(`PROBE1 intent-gap: ${outcome}`);
    expect(true).toBe(true);
  });
});
